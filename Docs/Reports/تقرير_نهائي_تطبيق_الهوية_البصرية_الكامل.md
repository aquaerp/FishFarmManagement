### 1. إنشاء نموذج جديد باستخدام BaseForm

```csharp
using FishFarmManager.Forms;
using FishFarmManager.Services;
using System;
using System.Drawing;
using System.Windows.Forms;

namespace FishFarmManager.Forms
{
    public partial class ProductForm : AquaFarmBaseForm
    {
        public ProductForm()
        {
            InitializeComponent();
            SetupUI();
        }
        
        private void InitializeComponent()
        {
            this.Text = "إدارة المنتجات";
            this.Size = new Size(900, 700);
        }
        
        private void SetupUI()
        {
            // شريط العنوان (موحد ومتسق)
            var titleBar = CreateTitleBar("إدارة المنتجات");
            
            // منطقة المحتوى
            var contentPanel = CreateContentPanel();
            
            // الجدول
            var grid = CreateDataGridView(20, 20, 860, 500);
            grid.Columns.Add(new DataGridViewTextBoxColumn { 
                Name = "الاسم", 
                HeaderText = "اسم المنتج",
                Width = 200 
            });
            contentPanel.Controls.Add(grid);
            
            // الأزرار (موحدة من ThemeManager)
            var addBtn = ThemeManager.CreateSuccessButton("➕ إضافة", 120, 40);
            var editBtn = ThemeManager.CreateSecondaryButton("✏️ تعديل", 120, 40);
            var deleteBtn = ThemeManager.CreateDangerButton("🗑️ حذف", 120, 40);
            var closeBtn = ThemeManager.CreateSecondaryButton("إغلاق", 120, 40);
            
            addBtn.Click += AddButton_Click;
            editBtn.Click += EditButton_Click;
            deleteBtn.Click += DeleteButton_Click;
            closeBtn.Click += (s, e) => this.Close();
            
            var buttonBar = CreateButtonBar(closeBtn, deleteBtn, editBtn, addBtn);
            
            // إضافة للنموذج
            this.Controls.Add(titleBar);
            this.Controls.Add(contentPanel);
            this.Controls.Add(buttonBar);
        }
        
        private void AddButton_Click(object sender, EventArgs e)
        {
            ShowInfo("نموذج إضافة منتج جديد");
        }
        
        private void EditButton_Click(object sender, EventArgs e)
        {
            ShowInfo("نموذج تعديل منتج");
        }
        
        private void DeleteButton_Click(object sender, EventArgs e)
        {
            if (Confirm("هل تريد حذف المنتج المحدد؟"))
            {
                // حذف المنتج
                ShowSuccess("تم الحذف بنجاح!");
            }
        }
    }
}
```

public class AdvancedForm : AquaFarmBaseForm
{
    private DataGridView _grid;
    
    public AdvancedForm()
    {
        InitializeComponent();
        SetupUI();
        LoadData();
    }
    
    private void SetupUI()
    {
        var titleBar = CreateTitleBar("إدارة متقدمة");
        
        var content = CreateContentPanel();
        _grid = CreateDataGridView(20, 20, 1160, 600);
        _grid.Columns.Add(new DataGridViewTextBoxColumn { Name = "ID" });
        _grid.Columns.Add(new DataGridViewTextBoxColumn { Name = "الاسم" });
        content.Controls.Add(_grid);
        
        var addBtn = ThemeManager.CreateSuccessButton("➕ إضافة");
        var editBtn = ThemeManager.CreateSecondaryButton("✏️ تعديل");
        var deleteBtn = ThemeManager.CreateDangerButton("🗑️ حذف");
        var closeBtn = ThemeManager.CreateSecondaryButton("إغلاق");
        
        addBtn.Click += (s, e) => Add();
        editBtn.Click += (s, e) => Edit();
        deleteBtn.Click += (s, e) => Delete();
        closeBtn.Click += (s, e) => this.Close();
        
        var buttonBar = CreateButtonBar(closeBtn, deleteBtn, editBtn, addBtn);
        
        this.Controls.Add(titleBar);
        this.Controls.Add(content);
        this.Controls.Add(buttonBar);
    }
    
    private void Add()
    {
        ShowInfo("إضافة سجل جديد");
    }
    
    private void Edit()
    {
        if (_grid.SelectedRows.Count > 0)
            ShowInfo("تعديل السجل");
        else
            ShowWarning("الرجاء تحديد سجل للتعديل");
    }
    
    private void Delete()
    {
        if (_grid.SelectedRows.Count > 0)
        {
            if (Confirm("هل تريد حذف السجل المحدد؟"))
            {
                // حذف
                ShowSuccess("تم الحذف بنجاح!");
            }
        }
        else
        {
            ShowWarning("الرجاء تحديد سجل للحذف");
        }
    }
    
    private void LoadData()
    {
        // تحميل البيانات
    }
}

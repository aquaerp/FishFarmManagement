### استخدام AquaFarmBaseForm

```csharp
// ملف جديد: Forms/MyNewForm.cs
using FishFarmManager.Forms;
using FishFarmManager.Services;

namespace FishFarmManager.Forms
{
    public partial class MyNewForm : AquaFarmBaseForm
    {
        public MyNewForm()
        {
            InitializeComponent();
            SetupUI();
        }
        
        private void InitializeComponent()
        {
            this.Text = "نموذج جديد";
            this.Size = new Size(800, 600);
        }
        
        private void SetupUI()
        {
            // شريط العنوان
            var titleBar = CreateTitleBar("إدارة البيانات");
            
            // المحتوى
            var contentPanel = CreateContentPanel();
            var grid = CreateDataGridView(20, 20, 760, 400);
            contentPanel.Controls.Add(grid);
            
            // الأزرار
            var saveBtn = ThemeManager.CreateSuccessButton("حفظ", 120, 40);
            var cancelBtn = ThemeManager.CreateSecondaryButton("إلغاء", 120, 40);
            saveBtn.Click += SaveButton_Click;
            cancelBtn.Click += (s, e) => this.Close();
            
            var buttonBar = CreateButtonBar(saveBtn, cancelBtn);
            
            // إضافة للنموذج
            this.Controls.Add(titleBar);
            this.Controls.Add(contentPanel);
            this.Controls.Add(buttonBar);
        }
        
        private void SaveButton_Click(object sender, EventArgs e)
        {
            if (Confirm("هل تريد حفظ التغييرات؟"))
            {
                // حفظ البيانات
                ShowSuccess("تم الحفظ بنجاح!");
                this.Close();
            }
        }
    }
}
```

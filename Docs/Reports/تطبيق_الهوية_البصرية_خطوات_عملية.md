```csharp
        #region طرق تطبيق الثيم - Apply Theme Methods

        /// <summary>
        /// تطبيق الثيم على نموذج
        /// </summary>
        public static void ApplyTheme(Form form)
        {
            if (form == null) return;

            form.BackColor = NeutralLightGray;
            form.Font = MainFont;
            form.RightToLeft = RightToLeft.Yes;
            form.RightToLeftLayout = true;

            // تطبيق على جميع العناصر الفرعية
            ApplyThemeToControls(form.Controls);
        }

        /// <summary>
        /// تطبيق الثيم على مجموعة من العناصر
        /// </summary>
        private static void ApplyThemeToControls(Control.ControlCollection controls)
        {
            foreach (Control control in controls)
            {
                // تطبيق الخطوط
                if (control.Font.Name != "Consolas" && control.Font.Name != "Courier New")
                {
                    control.Font = MainFont;
                }

                // تطبيق خاص للأزرار
                if (control is Button button)
                {
                    ApplyButtonTheme(button);
                }

                // تطبيق خاص للـ DataGridView
                if (control is DataGridView grid)
                {
                    ApplyDataGridViewTheme(grid);
                }

                // تطبيق خاص للـ MenuStrip
                if (control is MenuStrip menu)
                {
                    ApplyMenuStripTheme(menu);
                }

                // تطبيق خاص للـ TabControl
                if (control is TabControl tab)
                {
                    ApplyTabControlTheme(tab);
                }

                // تطبيق على العناصر الفرعية بشكل تكراري
                if (control.HasChildren)
                {
                    ApplyThemeToControls(control.Controls);
                }
            }
        }

        /// <summary>
        /// تطبيق ثيم الأزرار
        /// </summary>
        private static void ApplyButtonTheme(Button button)
        {
            button.Font = ButtonFont;
            button.FlatStyle = FlatStyle.Flat;
            button.Cursor = Cursors.Hand;
            button.Height = Math.Max(button.Height, 35);

            // إذا لم يكن له لون مخصص، استخدم اللون الأساسي
            if (button.BackColor == SystemColors.Control || button.BackColor == Color.Empty)
            {
                button.BackColor = PrimaryDeepBlue;
                button.ForeColor = White;
            }

            // إضافة تأثيرات Hover
            button.MouseEnter += (s, e) =>
            {
                if (button.BackColor == PrimaryDeepBlue)
                    button.BackColor = PrimaryLightBlue;
            };

            button.MouseLeave += (s, e) =>
            {
                if (button.BackColor == PrimaryLightBlue)
                    button.BackColor = PrimaryDeepBlue;
            };
        }

        /// <summary>
        /// تطبيق ثيم DataGridView
        /// </summary>
        private static void ApplyDataGridViewTheme(DataGridView grid)
        {
            grid.BackgroundColor = White;
            grid.BorderStyle = BorderStyle.None;
            grid.CellBorderStyle = DataGridViewCellBorderStyle.SingleHorizontal;
            grid.GridColor = BorderColor;
            grid.AlternatingRowsDefaultCellStyle.BackColor = GridAlternateRow;
            grid.DefaultCellStyle.SelectionBackColor = SecondarySkyBlue;
            grid.DefaultCellStyle.SelectionForeColor = White;
            grid.ColumnHeadersDefaultCellStyle.BackColor = PrimaryDeepBlue;
            grid.ColumnHeadersDefaultCellStyle.ForeColor = White;
            grid.ColumnHeadersDefaultCellStyle.Font = BoldFont;
            grid.EnableHeadersVisualStyles = false;
            grid.RowHeadersVisible = false;
            grid.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill;
            grid.SelectionMode = DataGridViewSelectionMode.FullRowSelect;
            grid.MultiSelect = false;
            grid.Font = MainFont;
        }

        /// <summary>
        /// تطبيق ثيم MenuStrip
        /// </summary>
        private static void ApplyMenuStripTheme(MenuStrip menu)
        {
            menu.BackColor = PrimaryDeepBlue;
            menu.ForeColor = White;
            menu.Font = MainFont;

            foreach (ToolStripMenuItem item in menu.Items)
            {
                item.BackColor = PrimaryDeepBlue;
                item.ForeColor = White;
            }
        }

        /// <summary>
        /// تطبيق ثيم TabControl
        /// </summary>
        private static void ApplyTabControlTheme(TabControl tab)
        {
            tab.Font = MainFont;
            
            foreach (TabPage page in tab.TabPages)
            {
                page.BackColor = NeutralLightGray;
                page.Font = MainFont;
            }
        }
```

### الخطوة 3.1: إنشاء ملف جديد Forms/AquaFarmBaseForm.cs

```csharp
using System;
using System.Drawing;
using System.Windows.Forms;
using FishFarmManager.Services;

namespace FishFarmManager.Forms
{
    /// <summary>
    /// نموذج أساسي لجميع نماذج AquaFarm Pro
    /// يطبق الهوية البصرية تلقائياً
    /// </summary>
    public class AquaFarmBaseForm : Form
    {
        protected AquaFarmBaseForm()
        {
            InitializeBaseForm();
        }

        private void InitializeBaseForm()
        {
            // تطبيق الإعدادات الأساسية
            this.RightToLeft = RightToLeft.Yes;
            this.RightToLeftLayout = true;
            this.BackColor = ThemeManager.NeutralLightGray;
            this.Font = ThemeManager.MainFont;
            this.StartPosition = FormStartPosition.CenterScreen;

            // تطبيق الثيم
            this.Load += (s, e) =>
            {
                try
                {
                    ThemeManager.ApplyTheme(this);
                }
                catch (Exception ex)
                {
                    System.Diagnostics.Debug.WriteLine($"خطأ في تطبيق الثيم: {ex.Message}");
                }
            };
        }

        /// <summary>
        /// إنشاء شريط عنوان موحد
        /// </summary>
        protected Panel CreateTitleBar(string title)
        {
            var panel = new Panel
            {
                Dock = DockStyle.Top,
                Height = 60,
                BackColor = ThemeManager.PrimaryDeepBlue
            };

            var label = new Label
            {
                Text = title,
                Font = ThemeManager.TitleFont,
                ForeColor = ThemeManager.White,
                AutoSize = false,
                Size = new Size(panel.Width - 40, 60),
                Location = new Point(20, 0),
                TextAlign = ContentAlignment.MiddleRight,
                Anchor = AnchorStyles.Top | AnchorStyles.Right | AnchorStyles.Left
            };

            panel.Controls.Add(label);
            return panel;
        }

        /// <summary>
        /// إنشاء شريط أزرار موحد
        /// </summary>
        protected Panel CreateButtonBar(params Button[] buttons)
        {
            var panel = new Panel
            {

                button.Location = new Point(x - button.Width, 15);
                button.Anchor = AnchorStyles.Top | AnchorStyles.Right;
                panel.Controls.Add(button);
                x -= button.Width + 10;
            }

            return panel;
        }
```

### الخطوة 5.1: إنشاء fix-colors.ps1

```powershell
# fix-colors.ps1
# سكريبت لاستبدال الألوان القديمة بـ ThemeManager

$files = Get-ChildItem -Path "Forms\" -Filter "*.cs" -Recurse

foreach ($file in $files) {
    $content = Get-Content $file.FullName -Raw
    
    # استبدال الألوان
    $content = $content -replace 'Color\.FromArgb\(0, 122, 204\)', 'ThemeManager.SecondarySkyBlue'
    $content = $content -replace 'Color\.FromArgb\(52, 152, 219\)', 'ThemeManager.SecondarySkyBlue'
    $content = $content -replace 'Color\.FromArgb\(46, 204, 113\)', 'ThemeManager.SuccessGreen'
    $content = $content -replace 'Color\.FromArgb\(0, 153, 204\)', 'ThemeManager.SecondarySkyBlue'
    $content = $content -replace 'Color\.FromArgb\(231, 76, 60\)', 'ThemeManager.ErrorRed'
    $content = $content -replace 'Color\.FromArgb\(155, 89, 182\)', 'ThemeManager.SecondarySkyBlue'
    $content = $content -replace 'Color\.FromArgb\(0, 51, 102\)', 'ThemeManager.PrimaryDeepBlue'
    
    # استبدال الخطوط
    $content = $content -replace 'new Font\("Segoe UI", 12, FontStyle\.Bold\)', 'ThemeManager.SubtitleFont'
    $content = $content -replace 'new Font\("Segoe UI", 10, FontStyle\.Bold\)', 'ThemeManager.ButtonFont'
    $content = $content -replace 'new Font\("Segoe UI", 10\)', 'ThemeManager.MainFont'
    $content = $content -replace 'new Font\("Segoe UI", 9\)', 'ThemeManager.SmallFont'
    
    # حفظ التغييرات
    Set-Content -Path $file.FullName -Value $content -Encoding UTF8
    
    Write-Host "✅ تم تحديث: $($file.Name)" -ForegroundColor Green
}

Write-Host "`n🎉 اكتمل الاستبدال التلقائي!" -ForegroundColor Cyan
```

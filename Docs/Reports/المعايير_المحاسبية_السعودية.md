### 1. المعايير الواجب تطبيقها

#### أ) معيار العرض والإفصاح (SOCPA Standard 1)

**المطلوب في النظام:**

#### ب) معيار المخزون (SOCPA Standard 2)

**المطلوب:**

- تقييم المخزون بالتكلفة أو صافي القيمة القابلة للتحقق أيهما أقل
- طريقة FIFO أو المتوسط المرجح
- إفصاح عن طريقة التقييم

```csharp
public class InventoryValuation
{
    public int Id { get; set; }
    public int InventoryItemId { get; set; }
    public InventoryItem InventoryItem { get; set; } = null!;
    
    public DateTime ValuationDate { get; set; }
    
    // طريقة التقييم (SOCPA)
    public InventoryValuationMethod Method { get; set; }
    
    // التكلفة
    [Column(TypeName = "decimal(18,2)")]
    public decimal CostValue { get; set; }
    
    // صافي القيمة القابلة للتحقق (NRV)
    [Column(TypeName = "decimal(18,2)")]
    public decimal NetRealizableValue { get; set; }
    
    // القيمة الدفترية (أقل من التكلفة و NRV)
    [Column(TypeName = "decimal(18,2)")]
    public decimal CarryingValue { get; set; }
    
    // مخصص هبوط القيمة
    [Column(TypeName = "decimal(18,2)")]
    public decimal ImpairmentProvision { get; set; }
}

public enum InventoryValuationMethod
{
    FIFO = 1,                   // الوارد أولاً صادر أولاً
    WeightedAverage = 2,        // المتوسط المرجح
    SpecificIdentification = 3  // التمييز الخاص
}
```

public class VATConfiguration
{
    public int Id { get; set; }

    [Required]
    [StringLength(15)]
    public string VATNumber { get; set; } = string.Empty; // الرقم الضريبي (15 رقم)

    public string InputVATAccountCode { get; set; } = string.Empty;  // ض.ق.م على المشتريات

{

    Annual = 3      // سنوي (للمنشآت الصغيرة)

public class PenaltyCalculator
{
    // غرامات ضريبة القيمة المضافة
    public static decimal CalculateVATPenalty(decimal taxAmount, int daysLate)
    {
        // 5% من قيمة الضريبة عن التأخير في التقديم
        var latePenalty = taxAmount * 0.05m;
        
        // 5% عن التأخير في السداد لمدة تصل إلى 30 يوم
        // + 10% إضافية عن كل 30 يوم تأخير إضافي
        var paymentPenalty = 0m;
        if (daysLate > 0 && daysLate <= 30)
        {
            paymentPenalty = taxAmount * 0.05m;
        }
        else if (daysLate > 30)
        {
            var periods = (int)Math.Ceiling((daysLate - 30) / 30.0);
            paymentPenalty = taxAmount * (0.05m + (periods * 0.10m));
            paymentPenalty = Math.Min(paymentPenalty, taxAmount); // الحد الأقصى 100%
        }
        
        return latePenalty + paymentPenalty;
    }
    
    // غرامات التأمينات الاجتماعية
    public static decimal CalculateGOSIPenalty(decimal contributionAmount, int daysLate)
    {
        // 2% شهرياً من المبلغ المستحق
        var months = Math.Ceiling(daysLate / 30.0);
        return contributionAmount * 0.02m * (decimal)months;
    }
    
    // غرامات مخالفات العمل
    public static decimal GetLaborViolationPenalty(LaborViolationType type)
    {
        return type switch
        {
            LaborViolationType.LateWages => 3000m, // 3000 ريال عن كل عامل
            LaborViolationType.NoContract => 5000m, // 5000 ريال عن كل عامل
            LaborViolationType.NoWorkPermit => 10000m, // 10000 ريال
            LaborViolationType.NoGOSI => 10000m, // 10000 ريال
            _ => 0m
        };
    }
}

public enum LaborViolationType
{
    LateWages,          // تأخير صرف الرواتب
    NoContract,         // عدم وجود عقد
    NoWorkPermit,       // عدم وجود رخصة عمل
    NoGOSI,             // عدم تسجيل في التأمينات
    UnsafeWorkplace     // بيئة عمل غير آمنة
}

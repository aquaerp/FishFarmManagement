```csharp
    {
        var cycles = _context.ProductionCycles
            .Where(c => c.Status == CycleStatus.Completed)
            .ToList();
            
        var costAnalysis = cycles.Select(cycle =>
        {
            var totalCost = _context.CostRecords
                .Where(c => c.ProductionCycleId == cycle.Id)
                .Sum(c => c.Amount);

                CostPerKg = costPerKg,
```

```csharp
public class FinancialService
{
    private readonly FishFarmContext _context;
    
    public FinancialService(FishFarmContext context)
    {
        _context = context;
    }
    
    // قائمة الدخل
    public IncomeStatementData GetIncomeStatement(DateTime fromDate, DateTime toDate)
    {
        // الإيرادات
        var revenue = _context.SalesOrders
            .Where(so => so.OrderDate >= fromDate && so.OrderDate <= toDate 
                      && so.Status == SalesOrderStatus.Delivered)
            .Sum(so => so.TotalAmount);
        
        // تكلفة البضاعة المباعة (COGS)
        var directCosts = _context.CostRecords
            .Where(c => c.Date >= fromDate && c.Date <= toDate 
                     && (c.Category == CostCategory.Fingerlings 
                      || c.Category == CostCategory.Feed 
                      || c.Category == CostCategory.Labor_Production))
            .Sum(c => c.Amount);
        
        var grossProfit = revenue - directCosts;
        var grossMargin = revenue > 0 ? (grossProfit / revenue) * 100 : 0;
        
        // المصروفات التشغيلية
        var operatingExpenses = _context.CostRecords
            .Where(c => c.Date >= fromDate && c.Date <= toDate 
                     && c.Category != CostCategory.Fingerlings 
                     && c.Category != CostCategory.Feed 
                     && c.Category != CostCategory.Labor_Production)
            .Sum(c => c.Amount);
        
        var netProfit = grossProfit - operatingExpenses;
        var netMargin = revenue > 0 ? (netProfit / revenue) * 100 : 0;
        
        return new IncomeStatementData
        {
            FromDate = fromDate,
            ToDate = toDate,
            Revenue = revenue,
            DirectCosts = directCosts,
            GrossProfit = grossProfit,
            GrossMargin = grossMargin,
            OperatingExpenses = operatingExpenses,
            NetProfit = netProfit,
            NetMargin = netMargin
        };
    }
    
    // تحليل الربحية حسب الدورة
    public List<CycleProfitability> GetCycleProfitability()
    {
        var completedCycles = _context.ProductionCycles
            .Where(c => c.Status == CycleStatus.Completed)
            .ToList();
        
        return completedCycles.Select(cycle =>
        {
            // الإيرادات
            var revenue = _context.SalesOrders
                .Where(so => so.Items.Any(i => i.ProductionCycleId == cycle.Id))
                .Sum(so => so.TotalAmount);
            
            // التكاليف
            var costs = _context.CostRecords
                .Where(c => c.ProductionCycleId == cycle.Id)
                .Sum(c => c.Amount);
            
            var profit = revenue - costs;
            var roi = costs > 0 ? (profit / costs) * 100 : 0;
            var profitMargin = revenue > 0 ? (profit / revenue) * 100 : 0;
            
            return new CycleProfitability
            {
                CycleId = cycle.Id,
                CycleName = cycle.Name,
                StartDate = cycle.StartDate,
                EndDate = cycle.EndDate,
                Revenue = revenue,
                TotalCosts = costs,
                Profit = profit,
                ProfitMargin = profitMargin,
                ROI = roi,
                Production = cycle.TotalHarvestWeight ?? 0,
                CostPerKg = cycle.TotalHarvestWeight > 0 
                    ? costs / (decimal)cycle.TotalHarvestWeight : 0
            };
        }).ToList();
    }
    
    // KPIs المالية
    public FinancialKPIs GetFinancialKPIs(DateTime? fromDate = null, DateTime? toDate = null)
    {
        fromDate ??= DateTime.Now.AddMonths(-12);
        toDate ??= DateTime.Now;
        
        // حساب مؤشرات الأداء
        var totalRevenue = _context.SalesOrders
            .Where(so => so.OrderDate >= fromDate && so.OrderDate <= toDate)
            .Sum(so => so.TotalAmount);
        
        var totalCosts = _context.CostRecords
            .Where(c => c.Date >= fromDate && c.Date <= toDate)
            .Sum(c => c.Amount);
        
        var netProfit = totalRevenue - totalCosts;
        
        return new FinancialKPIs
        {
            TotalRevenue = totalRevenue,
            TotalCosts = totalCosts,
            NetProfit = netProfit,
            ProfitMargin = totalRevenue > 0 ? (netProfit / totalRevenue) * 100 : 0,
            AverageSalesPerOrder = 0, // حساب
            AverageCostPerKg = 0,     // حساب
            CurrentAssets = 0,         // من المخزون + الذمم
            CurrentLiabilities = 0,    // الذمم الدائنة
            WorkingCapital = 0,
            QuickRatio = 0
        };
    }
}
```

```csharp
public class IncomeStatementForm : Form
```

```powershell

$backupPath = "E:\Fish Management\Backups\Backup_$backupDate"
```
